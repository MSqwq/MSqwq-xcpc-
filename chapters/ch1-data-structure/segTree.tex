\section{线段树模板}
\par \noindent 本章统一使用 1-indexed 区间 $[l,r]$。代码片段中的 $N$、$a$、$mod$、$inf$ 等常量按题目约束定义；查询函数的返回值必须使用对应的单位元。


\subsection{线段树：单点修改、区间查询}

\begin{minted}{c++}
struct SegPoint
{
    int sum[N<<2],mx[N<<2],mn[N<<2];
    void pushup(int p)
    {
        sum[p]=sum[p<<1]+sum[p<<1|1];
        mx[p]=max(mx[p<<1],mx[p<<1|1]);
        mn[p]=min(mn[p<<1],mn[p<<1|1]);
    }
    void build(int p,int l,int r)
    {
        if(l==r){sum[p]=mx[p]=mn[p]=a[l];return;}
        int mid=(l+r)>>1;
        build(p<<1,l,mid);build(p<<1|1,mid+1,r);pushup(p);
    }
    void modify(int p,int l,int r,int x,int v)
    {
        if(l==r){sum[p]=mx[p]=mn[p]=v;return;}
        int mid=(l+r)>>1;
        if(x<=mid)modify(p<<1,l,mid,x,v);
        else modify(p<<1|1,mid+1,r,x,v);
        pushup(p);
    }
    int query_sum(int p,int l,int r,int ql,int qr)
    {
        if(ql<=l&&r<=qr)return sum[p];
        int mid=(l+r)>>1,ans=0;
        if(ql<=mid)ans+=query_sum(p<<1,l,mid,ql,qr);
        if(qr>mid)ans+=query_sum(p<<1|1,mid+1,r,ql,qr);
        return ans;
    }
    int query_max(int p,int l,int r,int ql,int qr)
    {
        if(ql<=l&&r<=qr)return mx[p];
        int mid=(l+r)>>1,ans=-inf;
        if(ql<=mid)ans=max(ans,query_max(p<<1,l,mid,ql,qr));
        if(qr>mid)ans=max(ans,query_max(p<<1|1,mid+1,r,ql,qr));
        return ans;
    }
    int query_min(int p,int l,int r,int ql,int qr)
    {
        if(ql<=l&&r<=qr)return mn[p];
        int mid=(l+r)>>1,ans=inf;
        if(ql<=mid)ans=min(ans,query_min(p<<1,l,mid,ql,qr));
        if(qr>mid)ans=min(ans,query_min(p<<1|1,mid+1,r,ql,qr));
        return ans;
    }
}tr;
\end{minted}
单点加法可以把叶子的值改为原值加 v；区间查询的返回值要选择正确的单位元，求和用 0，求最大用负无穷，求最小用正无穷。

\subsection{线段树：区间加法与区间最值}

\begin{minted}{c++}
struct SegAdd
{
    int sum[N<<2],mx[N<<2],mn[N<<2],tag[N<<2];
    void apply(int p,int l,int r,int v)
    {
        sum[p]+=v*(r-l+1);
        mx[p]+=v;mn[p]+=v;tag[p]+=v;
    }
    void pushup(int p)
    {
        sum[p]=sum[p<<1]+sum[p<<1|1];
        mx[p]=max(mx[p<<1],mx[p<<1|1]);
        mn[p]=min(mn[p<<1],mn[p<<1|1]);
    }
    void pushdown(int p,int l,int r)
    {
        if(!tag[p]||l==r)return;
        int mid=(l+r)>>1;
        apply(p<<1,l,mid,tag[p]);
        apply(p<<1|1,mid+1,r,tag[p]);
        tag[p]=0;
    }
    void add(int p,int l,int r,int ql,int qr,int v)
    {
        if(ql<=l&&r<=qr){apply(p,l,r,v);return;}
        pushdown(p,l,r);int mid=(l+r)>>1;
        if(ql<=mid)add(p<<1,l,mid,ql,qr,v);
        if(qr>mid)add(p<<1|1,mid+1,r,ql,qr,v);
        pushup(p);
    }
};
\end{minted}
区间加会让最大值、最小值都加同一个数；这是因为区间内所有元素同时平移，元素的相对大小不变。

\subsection{线段树：区间加法和乘法}

\begin{minted}{c++}
struct SegMulAdd
{
    int sum[N<<2],add[N<<2],mul[N<<2];
    void apply(int p,int l,int r,int a,int m)
    {
        sum[p]=(sum[p]*m+a*(r-l+1))%mod;
        add[p]=(add[p]*m+a)%mod;
        mul[p]=mul[p]*m%mod;
    }
    void pushup(int p){sum[p]=(sum[p<<1]+sum[p<<1|1])%mod;}
    void pushdown(int p,int l,int r)
    {
        if(l==r)return;
        int mid=(l+r)>>1;
        apply(p<<1,l,mid,add[p],mul[p]);
        apply(p<<1|1,mid+1,r,add[p],mul[p]);
        add[p]=0;mul[p]=1;
    }
    void build(int p,int l,int r)
    {
        mul[p]=1;
        if(l==r){sum[p]=a[l]%mod;return;}
        int mid=(l+r)>>1;
        build(p<<1,l,mid);build(p<<1|1,mid+1,r);pushup(p);
    }
    void range_add(int p,int l,int r,int ql,int qr,int v)
    {
        if(ql<=l&&r<=qr){apply(p,l,r,v%mod,1);return;}
        pushdown(p,l,r);int mid=(l+r)>>1;
        if(ql<=mid)range_add(p<<1,l,mid,ql,qr,v);
        if(qr>mid)range_add(p<<1|1,mid+1,r,ql,qr,v);
        pushup(p);
    }
    void range_mul(int p,int l,int r,int ql,int qr,int v)
    {
        if(ql<=l&&r<=qr){apply(p,l,r,0,v%mod);return;}
        pushdown(p,l,r);int mid=(l+r)>>1;
        if(ql<=mid)range_mul(p<<1,l,mid,ql,qr,v);
        if(qr>mid)range_mul(p<<1|1,mid+1,r,ql,qr,v);
        pushup(p);
    }
    int query(int p,int l,int r,int ql,int qr)
    {
        if(ql<=l&&r<=qr)return sum[p];
        pushdown(p,l,r);int mid=(l+r)>>1,ans=0;
        if(ql<=mid)ans+=query(p<<1,l,mid,ql,qr);
        if(qr>mid)ans+=query(p<<1|1,mid+1,r,ql,qr);
        return ans%mod;
    }
}tr;
\end{minted}
标签代表变换 x 变为 x 乘 mul 加 add。下传父标签到子节点后，子节点旧的 add 必须先乘父 mul，再加父 add。

\subsection{线段树：区间赋值、区间推平}

\begin{minted}{c++}
struct SegSet
{
    int sum[N<<2],mx[N<<2],setv[N<<2];
    bool has[N<<2];
    void apply(int p,int l,int r,int v)
    {
        sum[p]=v*(r-l+1);mx[p]=v;setv[p]=v;has[p]=1;
    }
    void pushup(int p)
    {
        sum[p]=sum[p<<1]+sum[p<<1|1];
        mx[p]=max(mx[p<<1],mx[p<<1|1]);
    }
    void pushdown(int p,int l,int r)
    {
        if(!has[p]||l==r)return;
        int mid=(l+r)>>1;
        apply(p<<1,l,mid,setv[p]);
        apply(p<<1|1,mid+1,r,setv[p]);
        has[p]=0;
    }
    void assign(int p,int l,int r,int ql,int qr,int v)
    {
        if(ql<=l&&r<=qr){apply(p,l,r,v);return;}
        pushdown(p,l,r);int mid=(l+r)>>1;
        if(ql<=mid)assign(p<<1,l,mid,ql,qr,v);
        if(qr>mid)assign(p<<1|1,mid+1,r,ql,qr,v);
        pushup(p);
    }
};
\end{minted}
赋值操作覆盖之前所有加法和乘法标签。如果赋值可以为 0，必须用独立布尔变量 has 表示标签是否存在。

\subsection{线段树：区间取模}

操作通常是单点赋值、区间取模、区间求和。维护区间最大值 mx，若 mx 小于模数则整段不用递归；否则递归到叶子。

\begin{minted}{c++}
struct SegMod
{
    int sum[N<<2],mx[N<<2];
    void pushup(int p)
    {
        sum[p]=sum[p<<1]+sum[p<<1|1];
        mx[p]=max(mx[p<<1],mx[p<<1|1]);
    }
    void modify(int p,int l,int r,int x,int v)
    {
        if(l==r){sum[p]=mx[p]=v;return;}
        int mid=(l+r)>>1;
        if(x<=mid)modify(p<<1,l,mid,x,v);
        else modify(p<<1|1,mid+1,r,x,v);
        pushup(p);
    }
    void range_mod(int p,int l,int r,int ql,int qr,int v)
    {
        if(mx[p]<v)return;
        if(l==r){sum[p]=mx[p]%=v;return;}
        int mid=(l+r)>>1;
        if(ql<=mid)range_mod(p<<1,l,mid,ql,qr,v);
        if(qr>mid)range_mod(p<<1|1,mid+1,r,ql,qr,v);
        pushup(p);
    }
};
\end{minted}
均摊分析依赖 x mod v 后数值显著减小；如果题目还允许区间加，普通的最大值剪枝不够，需要更复杂的 Segment Tree Beats。

\subsection{线段树：区间异或和}

将整数拆成若干二进制位，每一位维护区间中 1 的个数。区间异或某个值时，值为 1 的位把 0 和 1 数量互换，值为 0 的位不动。

\begin{minted}{c++}
struct SegXor
{
    int one[31][N<<2];
    bool tag[31][N<<2];
    void apply(int b,int p,int l,int r)
    {
        one[b][p]=(r-l+1)-one[b][p];
        tag[b][p]^=1;
    }
    void pushdown(int b,int p,int l,int r)
    {
        if(!tag[b][p]||l==r)return;
        int mid=(l+r)>>1;
        apply(b,p<<1,l,mid);apply(b,p<<1|1,mid+1,r);
        tag[b][p]=0;
    }
    void change(int b,int p,int l,int r,int ql,int qr)
    {
        if(ql<=l&&r<=qr){apply(b,p,l,r);return;}
        pushdown(b,p,l,r);int mid=(l+r)>>1;
        if(ql<=mid)change(b,p<<1,l,mid,ql,qr);
        if(qr>mid)change(b,p<<1|1,mid+1,r,ql,qr);
        one[b][p]=one[b][p<<1]+one[b][p<<1|1];
    }
};
\end{minted}
查询区间和时把每一位的 1 的数量乘以 2 的 b 次方再相加。若值域到 1e18，位数开到 62。

\subsection{线段树：拆位维护与按位运算}

区间按位与和按位或也可以拆位。对某一位：

\par \noindent 区间 AND 0：该位全部变成 0。
\par \noindent 区间 OR 1：该位全部变成 1。
\par \noindent 区间 XOR 1：该位翻转。
\par \noindent 若区间该位已经全为 0 或全为 1，相关操作可以直接停止递归。

同时存在 AND、OR、XOR 时，需要把对每一位的变换表示成二元布尔函数，并在懒标记合并时复合函数；不要简单地把三个标记相加。

\subsection{线段树：最大子段和}

节点维护 sum、lmx、rmx、mx。最大子段要么完全在左儿子、完全在右儿子、要么横跨中点。

\begin{minted}{c++}
struct Info
{
    int sum,lmx,rmx,mx;
};
Info merge_info(Info a,Info b)
{
    Info c;
    c.sum=a.sum+b.sum;
    c.lmx=max(a.lmx,a.sum+b.lmx);
    c.rmx=max(b.rmx,b.sum+a.rmx);
    c.mx=max({a.mx,b.mx,a.rmx+b.lmx});
    return c;
}
\end{minted}
空区间返回时要使用带负无穷的单位元，不能直接返回全 0，否则全负数数组会得到错误答案。

\subsection{线段树：区间第一个满足条件的位置}

维护区间最大值或最小值，从根向下优先访问可能存在答案的儿子。

\begin{minted}{c++}
int first_pos(int p,int l,int r,int v)
{
    if(mx[p]<v)return -1;
    if(l==r)return l;
    int mid=(l+r)>>1;
    if(mx[p<<1]>=v)return first_pos(p<<1,l,mid,v);
    return first_pos(p<<1|1,mid+1,r,v);
}
\end{minted}
\subsection{动态开点线段树}

值域很大、修改点很少时使用。空节点不创建，空间复杂度与实际访问过的节点数成正比。

\begin{minted}{c++}
int ls[N*40],rs[N*40],sum[N*40],tot;
void add(int &p,int l,int r,int x,int v)
{
    if(!p)p=++tot;
    sum[p]+=v;
    if(l==r)return;
    int mid=(l+r)>>1;
    if(x<=mid)add(ls[p],l,mid,x,v);
    else add(rs[p],mid+1,r,x,v);
}
\end{minted}
\subsection{线段树合并}

两棵权值线段树合并时，空树直接返回另一棵，非空节点递归合并并相加信息。常用于树上每个节点维护颜色集合或子树统计。

\begin{minted}{c++}
int merge_tree(int x,int y,int l,int r)
{
    if(!x||!y)return x|y;
    cnt[x]+=cnt[y];
    if(l==r)return x;
    int mid=(l+r)>>1;
    ls[x]=merge_tree(ls[x],ls[y],l,mid);
    rs[x]=merge_tree(rs[x],rs[y],mid+1,r);
    return x;
}
\end{minted}
\subsection{Segment Tree Beats}

支持区间加、区间取最小、区间取最大、区间求和。节点必须维护最大值、次大值、最大值出现次数、最小值、次小值、最小值出现次数和区间和。

当 chmin 的新值大于次大值且小于最大值时，只有最大值这一类数会改变，可以整段打标；否则必须递归。chmax 同理。实现时最容易错的是 pushdown 中先恢复父节点对儿子的 max1/min1 约束，再传 add。比赛前必须用随机暴力对拍。

\subsection{扫描线线段树}

矩形面积并把每个矩形拆成两个事件：左边界加入，右边界删除。y 坐标离散成基本区间，线段树维护 cover 覆盖次数和 len 覆盖长度。相邻事件面积增加为 len 乘当前 x 差。矩形周长还需维护覆盖段数、左右端覆盖情况和横向变化量。

\subsection{线段树调试}

调试时可以递归输出每个节点的区间、sum、最值和懒标记；对拍时用长度小的随机数组，同时暴力执行每个操作。最常见错误是忘记 pushdown、赋值标签 0 与空标签混淆、乘法标签初始不是 1、查询后没有恢复节点信息。



\section{线段树进阶专题}
\subsection{历史版本和}
\par \noindent 给定一个长度为 $n$ 的序列 $a_i$，现在有 $m$ 次操作，操作有两种：
\begin{tcolorbox}
\begin{itemize}
\item 给定区间 $[l,r]$ 和 $v$，将区间 $[l,r]$ 加上 $v$
\item 给定区间 $[l,r]$，求区间 $[l,r]$ 的历史版本和
\end{itemize}
\end{tcolorbox}
\par \noindent 用线段树维护区间和 $sum$、区间历史和 $sumh$，但是我们这里不引入时间的概念，我们将历史和 $sumh$ 看成是 $sum$ 乘上另一个标记 $tag$，而每段时间，我们手动更新 $tag$.
\par \noindent 这里认为 $tag$ 标记优先级高于加标记：
\begin{minted}{c++}
// U216697 线段树区间历史版本和 
// https://www.luogu.com.cn/problem/U216697
const int N=100008;
int a[N],n,m;

struct Node
{
    int len;
    int sum,add;    //     区间和、 区间加标记
    int sumh,addh;  // 区间历史和、历史和加标记
    int tag;        // 次数乘上原来儿子上存的值
}t[N<<2];

void pushup(int u)
{
    t[u].sum=t[u<<1].sum+t[u<<1|1].sum;
    t[u].sumh=t[u<<1].sumh+t[u<<1|1].sumh;
}

// 一起下传 考虑标记之间影响
void puttag(int u,int add,int addh,int tag)
{
    t[u].sumh+=tag*t[u].sum+addh*t[u].len;
    t[u].sum+=t[u].len*add;
    
    t[u].addh+=tag*t[u].add+addh;
    t[u].tag+=tag;
    t[u].add+=add;
}

void mark_tag(int u,int tag)
{
    t[u].sumh+=tag*t[u].sum;
    t[u].addh+=tag*t[u].add;
    t[u].tag+=tag;
}
void mark_adh(int u,int addh)
{
    t[u].sumh+=addh*t[u].len;
    t[u].addh+=addh;
}
void mark_add(int u,int add)
{
    t[u].sum+=t[u].len*add;
    t[u].add+=add;
}
void pushdown(int u)
{
    if(t[u].tag)
    {
        mark_tag(u<<1,t[u].tag);
        mark_tag(u<<1|1,t[u].tag);
        t[u].tag=0;
    }
    if(t[u].addh)
    {
        mark_adh(u<<1,t[u].addh);
        mark_adh(u<<1|1,t[u].addh);
        t[u].addh=0;
    }
    if(t[u].add)
    {
        mark_add(u<<1,t[u].add);
        mark_add(u<<1|1,t[u].add);
        t[u].add=0;
    }
}
void build(int u,int l,int r)
{
    t[u].len=r-l+1;
    if(l==r)
    {
        t[u].sum=a[l];
        t[u].sumh=a[l];
        return;
    }
    int mid=(l+r)>>1;
    build(u<<1,l,mid);
    build(u<<1|1,mid+1,r);
    pushup(u);
}
void update(int u,int l,int r,int L,int R,int v)
{
    if(l>=L&&r<=R)
    {
        mark_add(u,v);
        return;
    }
    pushdown(u);
    int mid=(l+r)>>1;
    if(L<=mid)update(u<<1,l,mid,L,R,v);
    if(R>mid)update(u<<1|1,mid+1,r,L,R,v);
    pushup(u);
}
int query(int u,int l,int r,int L,int R)
{
    if(l>=L&&r<=R) 
        return t[u].sumh;
    pushdown(u);
    int mid=(l+r)>>1;
    int v=0;
    if(L<=mid)v+=query(u<<1,l,mid,L,R);
    if(R>mid)v+=query(u<<1|1,mid+1,r,L,R);
    return v;
}
signed main()
{
    scanf("%d%d",&n,&m);
    for(int i=1;i<=n;i++)scanf("%d",&a[i]);
    build(1,1,n);
    int op,x,y,l,r;
    for(int i=1;i<=m;i++)
    {
        scanf("%d",&op);
        if(op==1)
        {
            scanf("%d%d%d",&l,&r,&x);
            update(1,1,n,l,r,x);
        }
        else
        {
            scanf("%d%d",&x,&y);
            printf("%lld\n",query(1,1,n,x,y));
        }
        mark_tag(1,1);
    }
    return 0;
}
\end{minted}
\begin{tcolorbox}
\par \noindent 给出一个长度为 $n$ 的序列 $a$，每次询问一个区间 $[l,r]$。询问有多少个子区间 $[i,j]$ 满足子区间内不同的数的数量是奇数。
\end{tcolorbox}
\begin{itemize}
\item 将询问离线，从左往右扫描右端点 $r$
\item 线段树每个点 $i$ 维护的是 $[i,r]$ 这个段区间不同数的个数是偶数/奇数
\item 那么对于区间询问 $[L, R]$ 得到的结果： 有多少形如 $[(L\to R), r]$ 区间不同数的个数是 偶数/奇数
\end{itemize}
\textbf{注意：$[(L\to R), r]$ 的区间左端点在 $[L,R]$ 右端点始终是目前扫描到的 $r$ }
~\\
\par \noindent 对于维护 $[L,R]$ 子区间 显然是维护所有答案的历史版本和！
~\\

\par \noindent \textbf{有时维护很多标记时，标记之间的关系非常混乱，考虑是否能用矩阵的形式代替。}
~\\
\par \noindent 考虑维护：

\begin{itemize}
\item 区间长度：$len=r-l+1$
\item 区间和：$sum$
\item 历史区间和：$sumh$
\end{itemize}

\par \noindent 考虑每次区间加 $val$，那么考虑各个数字的变化：

\begin{itemize}
\item 对于区间和变化量： $len×val$
\item 对于区间历史和变化量：$sum+len×val$
\end{itemize}

\par \noindent 于是有：

\begin{itemize}
\item $len = len$
\item $sum = sum + len × val$
\item $sumh = sumh + sum + len × val$
\end{itemize}

{

        memset(m,0,sizeof(m));

        m[0][1]=m[0][2]=k;
        m[1][2]=1;

    Mat operator*(const Mat&B)const

        const Mat&A=*this;
        Mat ret;
        for(int k=0;k<M;k++)
            for(int i=0;i<M;i++)
                for(int j=0;j<M;j++)

};

        t[u].v.m[0][2]=a[l];
        return;

    if(R>mid)update(u<<1|1,mid+1,r,L,R,v);

    if(R>mid)ret=ret+query(u<<1|1,mid+1,r,L,R);
    return ret;
}

\subsection{区间 GCD}

\begin{tcolorbox}
给定一个长度为 N 的数列 A，以及 M 条指令，每条指令可能是以下两种之一：
\begin{itemize}
\item C l r d，表示把 $A[l],A[l+1],…,A[r]$ 都加上 $d$。
\item Q l r，表示询问 $A[l],A[l+1],…,A[r]$ 的最大公约数。
\end{itemize}
\end{tcolorbox}

$$
\operatorname{gcd}\left(a_1, a_2, \cdots, a_n\right)=\operatorname{gcd}\left(a_1, a_2-a_1, a_3-a_2, \cdots, a_n-a_{n-1}\right)
$$

\par \noindent 根据性质用 $b[i] = a[i] - a[i-1]$数组，表示原序列的差分序列，用线段数维护 $b[i]$ 的区间的最大公约数
\begin{itemize}
\item 询问 C l r d  等价于
\begin{enumerate}
\item $b[l]$ 加上 $d$、$b[r+1]$ 减去 $d$ ，只需两次线段数的单点修改即可
\item 对于原序列中 $a[i]$ 的值，需要支持区间加、点单查询，只需要用树状数组通过维护差分序列即可。
\end{enumerate}
\item 询问 Q l r 等于求出 $\operatorname{gcd}(a[l]$, query $(1, l+1, r))$
\end{itemize}
\begin{minted}{c++}
const int N=500010;

int n,m;
int a[N];
struct Node
{
    int l,r;
    int v;
}t[N<<2];
void pushup(int u)
{
    t[u].v=__gcd(t[u<<1].v,t[u<<1|1].v);
}
void build(int u,int l,int r)
{
    t[u]={l,r};
    if(l==r)
    {
        t[u].v=a[l]-a[l-1];
        return;
    }
    int mid=l+r>>1;
    build(u<<1,l,mid);
    build(u<<1|1,mid+1,r);
    pushup(u);
}
void modify(int u,int pos,int x)
{
    if(t[u].l==t[u].r)
    {
        t[u].v+=x;
        return;
    }
    int mid=t[u].l+t[u].r>>1;
    if(pos<=mid)modify(u<<1,pos,x);
    else modify(u<<1|1,pos,x);
    pushup(u);
}
int query(int u,int l,int r)
{
    if(l<=t[u].l&&t[u].r<=r)
    {
        return t[u].v;
    }
    int mid=t[u].l+t[u].r>>1;
    int v=0;
    if(l<=mid)v=__gcd(v,query(u<<1,l,r));
    if(r>mid)v=__gcd(v,query(u<<1|1,l,r));

}

有时候维护集合的 gcd 并支持：
\begin{enumerate}
\item 合并两个 gcd 集合
\item 对某些集合的所有数加上一个值
\end{enumerate}
$$
\operatorname{gcd}\left(a_1, a_2, \cdots, a_n\right)=\operatorname{gcd}\left(a_1, a_2-a_1, a_3-a_1, \cdots, a_n-a_{1}\right)
$$

\par \noindent 对于每一个集合维护两个数 
$$
f = a_1,g = \gcd(a_2-a_1, a_3-a_1, \cdots, a_n-a_{1})
$$
\par \noindent 对一个集合的所有数加上一个值操作 $f$ 即可。下面考虑如何合并两个集合：
~\\
\par \noindent 两个 gcd 合并时 需要将 $f_1$ 和 $f_2$ 合并 $g_1$ 和 $g_2$ 合并，令 $f_1$ 和 $f_2$ 合并成 $f_1$ （意味着对于两个序列合并，维护第一个序列的$a_1$）考虑差分的性质，对第二个集合来说等价于所有数减去 $f_1$，于是合并后集合应该有：
$$
f = f_1, g = \gcd(g_1, g_2,f_2 -f_1)
$$

        return x;
    }
    t[x].l=merge(t[x].l,t[y].l,l,mid);
    t[x].r=merge(t[x].r,t[y].r,mid+1,r);
    t[x].val=max(t[t[x].l].val,t[t[x].r].val);
    return x;
}
\end{minted}

\subsection{Segment Tree Beats!}
\par \noindent 区间最值操作往往采用以下办法 

\par \noindent 线段树维护：
\begin{itemize}
\item 区间最大值$\text{mx}$  
\item 区间严格次大值$\text {smx}$  
\item 区间和$\text{sum}$ 
\item 区间最大值个数$\text{cnt}$  
\item 区间最值懒标记$\text{lazy}$
\end{itemize}
\par \noindent 实现区间最小值操作，考虑u节点维护的区间，进行如下处理
\begin{itemize}
\item 当$\text{mx}\leq x$，显然这次修改不会对这个节点维护的区间产生影响，直接退出。
\item 当$\text{smx}<x<\text{mx}$，显然这次修改只会影响到这个区间所有的最大值，由此直接根据最大值个数更新区间和并且更新区间最大值并打上懒标记然后退出即可。
\item 当$\text{x}\leq \text{smx}$，无法直接更新于是递归左右子树。
\end{itemize}

\begin{minted}{c++}
struct Node
{
    int l,r; // 最大值 次大值 区间和
    int maxv,secmaxv,sum;
    int cnt;
    int tag;
}t[N<<2];
int n,m;
int a[N];
void pushup(int u)
{
    int x=u<<1,y=u<<1|1;
    t[u].sum=t[x].sum+t[y].sum;
    // 保证 t[x].maxv > t[u].maxv
    if(t[x].maxv<t[y].maxv)swap(x,y);
    
    if(t[x].maxv!=t[y].maxv)
    {
        t[u].maxv=t[x].maxv;
        t[u].secmaxv=max(t[x].secmaxv,t[y].maxv);
        t[u].cnt=t[x].cnt;
    }
    else
    {
        t[u].maxv=t[x].maxv;
        t[u].secmaxv=max(t[y].secmaxv,t[y].secmaxv);
        t[u].cnt=t[x].cnt+t[y].cnt;
    }
}
void puttag(int u,int x)
{
    if(t[u].maxv<=x)return;
    t[u].sum+=(x-t[u].maxv)*t[u].cnt;
    t[u].maxv=t[u].tag=x;
}
void pushdown(int u)
{
    if(t[u].tag==-1)return;
    puttag(u<<1,t[u].tag),puttag(u<<1|1,t[u].tag);
    t[u].tag=-1;
}
void build(int u,int l,int r)
{
    t[u]={l,r,0,-1,0,0,-1};
    if(l==r)
    {
        t[u].maxv=t[u].sum=a[l];
        t[u].cnt=1;
        return;
    }
    int mid=l+r>>1;
    build(u<<1,l,mid),build(u<<1|1,mid+1,r);
    pushup(u);
}
void modify(int u,int l,int r,int x) // 区间[l, r] a[i] = min(a[i], x)
{
    if(x>=t[u].maxv)return;
    // 当修改只会影响到区间最大值，并且不影响次大值时修改 意味着 secmaxv < x < maxv
    // 此时需要把区间最大值修改成 x
    if(t[u].l>=l&&t[u].r<=r&&t[u].secmaxv<x)
    {
        puttag(u,x);
        return;
    }
    pushdown(u);
    int mid=t[u].l+t[u].r>>1;
    if(l<=mid)modify(u<<1,l,r,x);
    if(r>mid)modify(u<<1|1,l,r,x);
    pushup(u);
}
int qmax(int u,int l,int r)
{
    if(t[u].l>=l&&t[u].r<=r)return t[u].maxv;
    int mid=t[u].r+t[u].l>>1;
    pushdown(u);
    int v=-1;
    if(l<=mid)v=max(v,qmax(u<<1,l,r));
    if(r>mid)v=max(v,qmax(u<<1|1,l,r));
    pushup(u);
    return v;
}
int qsum(int u,int l,int r)
{
    if(t[u].l>=l&&t[u].r<=r)
        return t[u].sum;
    int mid=t[u].r+t[u].l>>1;
    pushdown(u);
    int v=0;
    if(l<=mid)v+=qsum(u<<1,l,r);
    if(r>mid)v+=qsum(u<<1|1,l,r);
    pushup(u);
    return v;
}
\end{minted}

\par \noindent 维护一个长度为 $n$ 的序列，支持 $m$ 次操作，操作包括区间按位或一个数，区间按位与一个数，以及查询区间最大值。
\end{tcolorbox}

\par \noindent 线段树每个节点上维护区间与、区间或和区间最大值。
如果一次操作对区间与的影响和对区间或的影响相同，那么就说明对这整个区间的影响都是相同的，就是加上或减去同一个值，直接打标记即可，否则递归下去处理。
